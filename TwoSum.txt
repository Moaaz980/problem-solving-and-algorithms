// Soluzione brute force 

 twoSum(nums, target) {
        for (let i=0; i<nums.length; i++) {
            for (let j=i + 1; j<nums.length; j++) {
                if (nums[i] + nums[j] === target) {
                    return [i , j];
                }
            }
        }
    }    

// Soluzione efficiente con hashmap


twoSum(nums, target) {
        const map = new Map();
        for (let i=0; i<nums.length; i++) {
            // calcolo il complemento di ogni numero (quanto manca al numero per arrivare al target)
            const complement = target - nums[i];
            // controllo se ho già visto questo numero nella mappa 
            if (map.has(complement)) {
                return [map.get(complement) , i];
            }
            else {
                map.set(nums[i] , i);
            }
        }
    }
