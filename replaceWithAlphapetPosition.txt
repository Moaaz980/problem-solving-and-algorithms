function alphabetPosition(text) {
  const alphabetMap = new Map([
  ["a", 1],
  ["b", 2],
  ["c", 3],
  ["d", 4],
  ["e", 5],
  ["f", 6],
  ["g", 7],
  ["h", 8],
  ["i", 9],
  ["j", 10],
  ["k", 11],
  ["l", 12],
  ["m", 13],
  ["n", 14],
  ["o", 15],
  ["p", 16],
  ["q", 17],
  ["r", 18],
  ["s", 19],
  ["t", 20],
  ["u", 21],
  ["v", 22],
  ["w", 23],
  ["x", 24],
  ["y", 25],
  ["z", 26]
  ]);
  
  let res = '';
  let caratteri = Array.from(text).filter(char => /[a-zA-Z]+/.test(char)).join('');
  
  
  // scorro la stringa e per ogni carattere appendo la sua posizione e se è lultima iterazione non appendo lo spazio 
  
  for (let i=0; i<caratteri.length; i++) {
      const carattere = alphabetMap.get(caratteri.charAt(i).toLowerCase());
        res += i < caratteri.length -1 ? carattere  + ' ' : carattere;
  }
    return res;
  }
