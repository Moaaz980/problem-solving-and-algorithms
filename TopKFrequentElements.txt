class Solution {
    topKFrequent(arr , k) {
    const map = new Map() , res = new Set();
    for (let val of arr) {
        map.set(val , (map.get(val) + 1) || 1);
    }
    const orderedArray = Array.from(map.values()).sort((a , b) => b - a).slice(0 , k);
    
     for (let [key , value] of map.entries()) {
        for (let i=0; i<orderedArray.length; i++) {
            if (orderedArray[i] === value) {
                res.add(key);
            }
        }
     }
    return Array.from(res);
}
}
