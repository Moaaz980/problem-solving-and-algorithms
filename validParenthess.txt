    isValid(s) {
    const stack = [];
    const map = new Map(
        [
        [')' , '('] , 
        [']' , '['] , 
        ['}' , '{']    
        ]
        );
    
    for (let i=0; i<s.length; i++) {
        if (
            s.charAt(i) === '(' ||
            s.charAt(i) === '[' || 
            s.charAt(i) === '{' 
            {
                stack.push(s.charAt(i));
            }
            )
            if (
            s.charAt(i) === ')' ||
            s.charAt(i) === ']' || 
            s.charAt(i) === '}' 
                )
                {
                    if (stack[stack.length - 1] === map.get(s.charAt(i))) {
                        stack.pop();
                    }
                }
    }
    return stack.length === 0;
